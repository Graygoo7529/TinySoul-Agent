### 1.2 修改 ActivityGlimpse 组件

修改文件：`visualization/src/features/chat/ActivityGlimpse.tsx`

```typescript
interface ActivityGlimpseProps {
  state: ActivityState;
  message?: string;
  phase?: string;
  progress?: number;
}

export function ActivityGlimpse({ 
  state, 
  message, 
  phase, 
  progress 
}: ActivityGlimpseProps) {
  const colors = stateColors[state];
  
  return (
    <div className={`
      border-l-4 ${colors.border}
      ${colors.background}
      rounded-lg p-3
      transition-all duration-300
    `}>
      <div className="flex items-center gap-2">
        {/* 状态图标 */}
        <StatusIcon state={state} className={colors.icon} />
        
        {/* 状态文本 */}
        <div className="flex-1">
          <div className="text-sm font-medium text-fg-base">
            {message}
          </div>
          {phase && (
            <div className="text-xs text-fg-faint mt-0.5">
              {phase}
            </div>
          )}
        </div>
        
        {/* 进度指示（如果有） */}
        {progress !== undefined && (
          <div className="text-xs text-fg-faint">
            {Math.round(progress * 100)}%
          </div>
        )}
      </div>
      
      {/* 进度条 */}
      {progress !== undefined && (
        <div className="mt-2 h-1 bg-fg-faint/10 rounded-full overflow-hidden">
          <div 
            className="h-full bg-accent transition-all duration-300"
            style={{ width: `${progress * 100}%` }}
          />
        </div>
      )}
    </div>
  );
}

function StatusIcon({ state, className }: { state: ActivityState; className: string }) {
  const icons = {
    idle: <Circle size={16} />,
    thinking: <Brain size={16} className="animate-pulse" />,
    working: <Cpu size={16} className="animate-spin" />,
    waiting: <Clock size={16} />,
    suspended: <Pause size={16} />,
    error: <AlertCircle size={16} />
  };
  
  return <span className={className}>{icons[state]}</span>;
}
```

新增两种显示模式：

- **紧凑模式**（新增）：只显示关键信息

### 2.2 修改 ContextPane 组件

修改文件：`visualization/src/features/context/ContextPane.tsx`

```typescript
type DisplayMode = "detailed" | "compact";

export function ContextPane() {
  const [mode, setMode] = useState<DisplayMode>("compact");
  
  return (
    <div className="flex flex-col h-full">
      {/* 工具栏 */}
      <div className="flex items-center justify-between px-4 py-2 border-b border-fg-faint/10">
        <h2 className="text-sm font-medium text-fg-base">Context</h2>
        
        {/* 模式切换 */}
        <button
          onClick={() => setMode(m => m === "detailed" ? "compact" : "detailed")}
          className="p-1 hover:bg-surface-3 rounded transition-colors"
          title={mode === "detailed" ? "紧凑模式" : "详细模式"}
        >
          {mode === "detailed" ? <Minimize2 size={14} /> : <Maximize2 size={14} />}
        </button>
      </div>
      
      {/* 资源目录 */}
      <div className="flex-1 overflow-auto">
        <ResourceDirectory mode={mode} />
      </div>
    </div>
  );
}
```

### 2.3 创建资源目录组件

新建文件：`visualization/src/features/context/ResourceDirectory.tsx`

```typescript
interface ResourceDirectoryProps {
  mode: DisplayMode;
}

export function ResourceDirectory({ mode }: ResourceDirectoryProps) {
  const resources = useContextResources();  // 假设有这个 hook
  
  return (
    <div className="space-y-2 p-2">
      {resources.map(section => (
        <ResourceSection
          key={section.id}
          section={section}
          mode={mode}
        />
      ))}
    </div>
  );
}

function ResourceSection({ section, mode }: { section: ResourceSection; mode: DisplayMode }) {
  const [expanded, setExpanded] = useState(true);
  
  return (
    <div className="border border-fg-faint/10 rounded-lg overflow-hidden">
      {/* 区域头部 */}
      <button
        onClick={() => setExpanded(!expanded)}
        className="w-full flex items-center gap-2 px-3 py-2 bg-surface-2 hover:bg-surface-3 transition-colors"
      >
        <ChevronRight 
          size={14} 
          className={`transition-transform ${expanded ? "rotate-90" : ""}`}
        />
        <span className="text-xs font-medium text-fg-base flex-1 text-left">
          {section.title}
        </span>
        <span className="text-[10px] text-fg-faint">
          {section.items.length}
        </span>
      </button>
      
      {/* 资源列表 */}
      {expanded && (
        <div className={mode === "compact" ? "divide-y divide-fg-faint/5" : "space-y-1 p-1"}>
          {section.items.map(item => (
            <ResourceItem
              key={item.id}
              item={item}
              mode={mode}
            />
          ))}
        </div>
      )}
    </div>
  );
}

function ResourceItem({ item, mode }: { item: Resource; mode: DisplayMode }) {
  if (mode === "compact") {
    return (
      <div className="flex items-center gap-2 px-3 py-1.5 hover:bg-surface-2 transition-colors cursor-pointer">
        <FileIcon type={item.type} size={12} />
        <span className="text-xs text-fg-base truncate flex-1">
          {item.name}
        </span>
      </div>
    );
  }
  
  // 详细模式
  return (
    <div className="p-2 hover:bg-surface-2 rounded transition-colors cursor-pointer">
      <div className="flex items-center gap-2">
        <FileIcon type={item.type} size={14} />
        <span className="text-xs font-medium text-fg-base">
          {item.name}
        </span>
      </div>
      {item.description && (
        <div className="text-[10px] text-fg-faint mt-1 ml-5">
          {item.description}
        </div>
      )}
    </div>
  );
}
```
